cmake_minimum_required(VERSION 3.21)
project(UpdateServiceContracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Network)
add_executable(update_service_contracts main.cpp ../../src/ui/app/UpdateService.cpp)
target_include_directories(update_service_contracts PRIVATE ../../src/ui/app)
target_link_libraries(update_service_contracts PRIVATE Qt6::Core Qt6::Network)
if(MSVC)
  target_compile_options(update_service_contracts PRIVATE /utf-8 /W4)
  target_compile_definitions(update_service_contracts PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(update_service_contracts PRIVATE user32)
endif()
