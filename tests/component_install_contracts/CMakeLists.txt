cmake_minimum_required(VERSION 3.21)
project(CGPlayComponentInstallContracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core)
if(NOT CGPLAY_SOURCE_ROOT OR NOT CGPLAY_TEST_EXTRACTED_DIR)
  message(FATAL_ERROR "Pass CGPLAY_SOURCE_ROOT and CGPLAY_TEST_EXTRACTED_DIR explicitly")
endif()
add_executable(manifest_policy_probe manifest_policy.cpp)
target_include_directories(manifest_policy_probe PRIVATE "${CGPLAY_SOURCE_ROOT}/src")
target_link_libraries(manifest_policy_probe PRIVATE Qt6::Core)
add_executable(recording_runtime_probe recording_runtime.cpp)
target_include_directories(recording_runtime_probe PRIVATE "${CGPLAY_SOURCE_ROOT}/src")
target_link_libraries(recording_runtime_probe PRIVATE Qt6::Core)
add_executable(component_payload_probe main.cpp)
target_include_directories(component_payload_probe PRIVATE "${CGPLAY_SOURCE_ROOT}/src")
target_link_libraries(component_payload_probe PRIVATE Qt6::Core)
add_executable(export_error_storage_probe error_storage.cpp)
target_include_directories(export_error_storage_probe PRIVATE "${CGPLAY_TEST_EXTRACTED_DIR}")
target_link_libraries(export_error_storage_probe PRIVATE Qt6::Core)
if(WIN32)
  target_link_libraries(manifest_policy_probe PRIVATE user32)
  target_link_libraries(recording_runtime_probe PRIVATE user32)
  target_link_libraries(component_payload_probe PRIVATE user32)
  target_link_libraries(export_error_storage_probe PRIVATE user32)
endif()
if(MSVC)
  target_compile_options(manifest_policy_probe PRIVATE /utf-8)
  target_compile_options(recording_runtime_probe PRIVATE /utf-8)
  target_compile_options(component_payload_probe PRIVATE /utf-8)
  target_compile_options(export_error_storage_probe PRIVATE /utf-8)
endif()
