cmake_minimum_required(VERSION 3.21)
project(UpdateUICacheContracts LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Concurrent)
if(NOT EXISTS "${EXTRACTED_METHODS_DIR}/update_methods.inc")
    message(FATAL_ERROR "Generate the exact production method extraction with run.py first")
endif()
add_executable(update_ui_cache_contracts main.cpp)
target_compile_features(update_ui_cache_contracts PRIVATE cxx_std_17)
target_include_directories(update_ui_cache_contracts PRIVATE ../../src/ui/app "${EXTRACTED_METHODS_DIR}")
target_link_libraries(update_ui_cache_contracts PRIVATE Qt6::Core Qt6::Widgets Qt6::Concurrent)
if(MSVC)
    target_compile_options(update_ui_cache_contracts PRIVATE /utf-8 /W4)
    target_compile_definitions(update_ui_cache_contracts PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(update_ui_cache_contracts PRIVATE user32)
endif()
