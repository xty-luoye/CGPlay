cmake_minimum_required(VERSION 3.28)
project(CGPlayRecorder VERSION 1.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The recorder requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_C_STANDARD 17)
set(OBS_ROOT "E:/rjkf/CGPlay-deps/obs-33.0.0" CACHE PATH "Pinned private OBS root")
set(OBS_SOURCE "${OBS_ROOT}/upstream")
set(OBS_SDK "${OBS_ROOT}/sdk")
set(RECORDER_STAGE "${OBS_ROOT}/stage" CACHE PATH "Private runtime")
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS)
add_library(obs_sdk INTERFACE)
target_include_directories(obs_sdk INTERFACE "${OBS_SOURCE}/libobs" "${OBS_SOURCE}/deps/w32-pthreads" "${OBS_SDK}/include")
target_link_libraries(obs_sdk INTERFACE "${OBS_SDK}/lib/obs.lib")
add_executable(CGPlayRecorder main.cpp Protocol.cpp RuntimeManifest.cpp RecordingSession.cpp TestSources.cpp)
target_link_libraries(CGPlayRecorder PRIVATE obs_sdk bcrypt ole32 uuid dxgi user32 delayimp)
target_link_options(CGPlayRecorder PRIVATE /DELAYLOAD:obs.dll)
target_compile_options(CGPlayRecorder PRIVATE /utf-8 /W4 /EHsc)
set_target_properties(CGPlayRecorder PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${RECORDER_STAGE}" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${RECORDER_STAGE}")

# Compile only the two adapted capture modules against the exact official DLL.
set(WC "${OBS_SOURCE}/plugins/win-capture")
add_library(win-capture MODULE "${WC}/plugin-main.c" "${WC}/window-capture.c" "${WC}/duplicator-monitor-capture.c" "${WC}/monitor-capture.c" "${WC}/cursor-capture.c" "${WC}/dc-capture.c" "${WC}/audio-helpers.c")
target_include_directories(win-capture PRIVATE "${OBS_SOURCE}/libobs-winrt")
target_link_libraries(win-capture PRIVATE obs_sdk "${OBS_SDK}/lib/w32-pthreads.lib" user32 gdi32 dwmapi dxgi)
target_compile_definitions(win-capture PRIVATE OBS_VERSION="33.0.0")
set_target_properties(win-capture PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY_RELEASE "${RECORDER_STAGE}/core/win-capture" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${RECORDER_STAGE}/core/win-capture")

set(WA "${OBS_SOURCE}/plugins/win-wasapi")
add_library(win-wasapi MODULE "${WA}/win-wasapi.cpp" "${WA}/wasapi-notify.cpp" "${WA}/enum-wasapi.cpp" "${WA}/plugin-main.cpp")
target_link_libraries(win-wasapi PRIVATE obs_sdk avrt ole32 uuid user32)
target_compile_options(win-wasapi PRIVATE /utf-8 /EHsc)
set_target_properties(win-wasapi PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY_RELEASE "${RECORDER_STAGE}/core/win-wasapi" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${RECORDER_STAGE}/core/win-wasapi")
